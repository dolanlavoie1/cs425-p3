## Design

For /0, cidr_mask(0) returns 0 to avoid a 32 bit shift. The network is 0.0.0.0 and the broadcast is 255.255.255.255. The host count uses 64 bits to avoid an overflow. For /31, there are 2 usable addresses. cidr_first_host() returns the network address, and cidr_last_host() returns the broadcast address. For /32, there is only 1 host. The network, broadcast, first host, and last host are all the same address. 
For the longest-prefix lookup, the forwarding table stores routes in a growing array. table_lookup() then scans every route. It first checks if the destination is inside the route using the subnet mask. If it matches, it compares the route's prefix length with the best match found so far. It will select the route with the greatest prefix. For example, /25 would be selected over /24.

bool ip_parse(const char *text, uint32_t *addr)
{
  uint32_t value = 0;
  const char *cursor = text;

  if (text == NULL || addr == NULL) {
    return false;
  }

  for (unsigned int octet_index = 0; octet_index < 4; octet_index++) {
    unsigned int octet = 0;
    unsigned int digit_count = 0;

    if (*cursor < '0' || *cursor > '9') {
      return false;
    }
    if (*cursor == '0' && cursor[1] >= '0' && cursor[1] <= '9') {
      return false;
    }

    while (*cursor >= '0' && *cursor <= '9') {
      octet = octet * 10U + (unsigned int)(*cursor - '0');
      digit_count++;
      if (digit_count > 3 || octet > 255U) {
        return false;
      }
      cursor++;
    }

    value = (value << 8) | (uint32_t)octet;
    if (octet_index < 3) {
      if (*cursor != '.') {
        return false;
      }
      cursor++;
    } else if (*cursor != '\0') {
      return false;
    }
  }

  *addr = value;
  return true;
}

void ip_format(uint32_t addr, char out[IP_STR_LEN])
{
  (void)snprintf(out, IP_STR_LEN, "%u.%u.%u.%u",
                 (unsigned int)((addr >> 24) & 0xffU),
                 (unsigned int)((addr >> 16) & 0xffU),
                 (unsigned int)((addr >> 8) & 0xffU),
                 (unsigned int)(addr & 0xffU));
}

bool cidr_parse(const char *text, cidr_t *block)
{
  const char *slash;
  const char *prefix_text;
  size_t address_length;
  uint32_t address;
  unsigned int prefix = 0;
  unsigned int digit_count = 0;
  char address_text[IP_STR_LEN];

  if (text == NULL || block == NULL) {
    return false;
  }

  slash = text;
  while (*slash != '\0' && *slash != '/') {
    slash++;
  }
  if (*slash != '/') {
    return false;
  }

  address_length = (size_t)(slash - text);
  if (address_length == 0 || address_length >= IP_STR_LEN) {
    return false;
  }
  for (size_t index = 0; index < address_length; index++) {
    address_text[index] = text[index];
  }
  address_text[address_length] = '\0';
  if (!ip_parse(address_text, &address)) {
    return false;
  }

  prefix_text = slash + 1;
  if (*prefix_text == '\0' ||
      (*prefix_text == '0' && prefix_text[1] != '\0')) {
    return false;
  }
  while (*prefix_text >= '0' && *prefix_text <= '9') {
    prefix = prefix * 10U + (unsigned int)(*prefix_text - '0');
    digit_count++;
    if (digit_count > 2 || prefix > 32U) {
      return false;
    }
    prefix_text++;
  }
  if (*prefix_text != '\0') {
    return false;
  }

  block->addr = address;
  block->prefix = (uint8_t)prefix;
  return true;
}

void cidr_format(cidr_t block, char out[CIDR_STR_LEN])
{
  char address[IP_STR_LEN];
  unsigned int prefix = block.prefix > 32U ? 32U : (unsigned int)block.prefix;

  ip_format(block.addr, address);
  (void)snprintf(out, CIDR_STR_LEN, "%s/%u", address, prefix);
}

uint32_t cidr_mask(uint8_t prefix)
{
  if (prefix == 0) {
    return 0;
  }
  if (prefix >= 32) {
    return UINT32_MAX;
  }
  return UINT32_MAX << (32U - prefix);
}

uint32_t cidr_network(cidr_t block)
{
  return block.addr & cidr_mask(block.prefix);
}

uint32_t cidr_broadcast(cidr_t block)
{
  return cidr_network(block) | ~cidr_mask(block.prefix);
}

uint32_t cidr_first_host(cidr_t block)
{
  uint8_t prefix = block.prefix > 32 ? 32 : block.prefix;

  if (prefix >= 31) {
    return cidr_network(block);
  }
  return cidr_network(block) + 1U;
}

uint32_t cidr_last_host(cidr_t block)
{
  uint8_t prefix = block.prefix > 32 ? 32 : block.prefix;

  if (prefix >= 31) {
    return cidr_broadcast(block);
  }
  return cidr_broadcast(block) - 1U;
}

uint64_t cidr_host_count(cidr_t block)
{
  uint8_t prefix = block.prefix > 32 ? 32 : block.prefix;

  if (prefix == 31) {
    return 2;
  }
  if (prefix == 32) {
    return 1;
  }
  return (UINT64_C(1) << (32U - prefix)) - 2;
}

bool cidr_contains(cidr_t block, uint32_t addr)
{
  uint32_t mask = cidr_mask(block.prefix);

  return (addr & mask) == (block.addr & mask);
}

bool cidr_is_network(cidr_t block)
{
  return cidr_network(block) == block.addr;
}

bool cidr_subnet(cidr_t block, uint32_t n, uint32_t i, cidr_t *out)
{
  uint8_t prefix = block.prefix;
  uint32_t extra_bits = 0;
  uint32_t remaining;
  uint8_t new_prefix;
  cidr_t result;

  if (out == NULL || block.prefix > 32 || !cidr_is_network(block) || n == 0 ||
      (n & (n - 1U)) != 0 || i >= n) {
    return false;
  }

  remaining = n;
  while (remaining > 1) {
    remaining >>= 1;
    extra_bits++;
  }
  if (prefix > 32 || extra_bits > 32U - prefix) {
    return false;
  }

  new_prefix = (uint8_t)(prefix + extra_bits);
  result.prefix = new_prefix;
  result.addr = block.addr;
  if (i != 0 && new_prefix < 32) {
    result.addr += i << (32U - new_prefix);
  } else if (i != 0) {
    result.addr += i;
  }
  *out = result;
  return true;
}

bool route_parse(const char *line, route_t *route)
{
  const char *tokens[3] = {NULL, NULL, NULL};
  size_t lengths[3] = {0, 0, 0};
  size_t token_count = 0;
  const char *cursor;
  route_t parsed;
  char destination[CIDR_STR_LEN];
  char gateway[IP_STR_LEN];
  cidr_t destination_block;
  uint32_t gateway_address;

  if (line == NULL || route == NULL) {
    return false;
  }

  cursor = line;
  while (*cursor != '\0') {
    while (*cursor != '\0' && isspace((unsigned char)*cursor)) {
      cursor++;
    }
    if (*cursor == '\0') {
      break;
    }
    if (token_count >= 3) {
      return false;
    }
    tokens[token_count] = cursor;
    while (*cursor != '\0' && !isspace((unsigned char)*cursor)) {
      cursor++;
    }
    lengths[token_count] = (size_t)(cursor - tokens[token_count]);
    token_count++;
  }

  if (token_count != 3 || lengths[0] >= sizeof(destination) ||
      lengths[1] >= sizeof(gateway) || lengths[2] == 0 ||
      lengths[2] >= DEV_NAME_LEN) {
    return false;
  }

  memcpy(destination, tokens[0], lengths[0]);
  destination[lengths[0]] = '\0';
  memcpy(gateway, tokens[1], lengths[1]);
  gateway[lengths[1]] = '\0';

  if (strcmp(destination, "default") == 0) {
    destination_block.addr = 0;
    destination_block.prefix = 0;
  } else if (!cidr_parse(destination, &destination_block) ||
             !cidr_is_network(destination_block)) {
    return false;
  }

  if (strcmp(gateway, "direct") == 0) {
    gateway_address = 0;
  } else if (!ip_parse(gateway, &gateway_address)) {
    return false;
  }

  parsed.dest = destination_block;
  parsed.gateway = gateway_address;
  memcpy(parsed.dev, tokens[2], lengths[2]);
  parsed.dev[lengths[2]] = '\0';
  *route = parsed;
  return true;
}

bool table_add(fwd_table_t *table, const route_t *route)
{
  route_t *grown_routes;
  size_t new_capacity;

  if (table == NULL || route == NULL || route->dest.prefix > 32 ||
      !cidr_is_network(route->dest)) {
    return false;
  }
  for (size_t index = 0; index < table->count; index++) {
    if (table->routes[index].dest.addr == route->dest.addr &&
        table->routes[index].dest.prefix == route->dest.prefix) {
      return false;
    }
  }

  if (table->count == table->capacity) {
    new_capacity = table->capacity == 0 ? 4 : table->capacity * 2;
    if (new_capacity < table->capacity ||
        new_capacity > SIZE_MAX / sizeof(*table->routes)) {
      // GCOVR_EXCL_START
      return false;
      // GCOVR_EXCL_STOP
    }
    grown_routes = realloc(table->routes,
                           new_capacity * sizeof(*table->routes));
    if (grown_routes == NULL) {
      // GCOVR_EXCL_START
      return false;
      // GCOVR_EXCL_STOP
    }
    table->routes = grown_routes;
    table->capacity = new_capacity;
  }

  table->routes[table->count] = *route;
  table->count++;
  return true;
}

bool table_load(fwd_table_t *table, FILE *in, size_t *bad_line)
{
  char *line = NULL;
  size_t line_capacity = 0;
  size_t line_number = 0;
  ssize_t line_length;
  route_t route;
  char *comment;

  if (bad_line != NULL) {
    *bad_line = 0;
  }
  if (table == NULL || in == NULL) {
    return false;
  }

  while ((line_length = getline(&line, &line_capacity, in)) != -1) {
    (void)line_length;
    line_number++;
    comment = strchr(line, '#');
    if (comment != NULL) {
      *comment = '\0';
    }
    comment = line;
    while (*comment != '\0' && isspace((unsigned char)*comment)) {
      comment++;
    }
    if (*comment == '\0') {
      continue;
    }
    if (!route_parse(comment, &route) || !table_add(table, &route)) {
      if (bad_line != NULL) {
        *bad_line = line_number;
      }
      free(line);
      return false;
    }
  }

  free(line);
  if (ferror(in)) {
    return false;
  }
  return true;
}

const route_t *table_lookup(const fwd_table_t *table, uint32_t dest)
{
  const route_t *best = NULL;

  if (table == NULL) {
    return NULL;
  }
  for (size_t index = 0; index < table->count; index++) {
    const route_t *candidate = &table->routes[index];
    if (cidr_contains(candidate->dest, dest) &&
        (best == NULL || candidate->dest.prefix > best->dest.prefix)) {
      best = candidate;
    }
  }
  return best;
}

#ifndef LAB_H
#define LAB_H

#include <stdbool.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>

/*
 * Every address in this file is a uint32_t in host byte order, so the first
 * octet of 192.168.1.10 is the most significant byte: 0xC0A8010A.
 */

/** Room for the longest dotted quad, "255.255.255.255", and its '\0'. */
#define IP_STR_LEN 16

/** Room for the longest CIDR string, "255.255.255.255/32", and its '\0'. */
#define CIDR_STR_LEN 19

/** Room for an interface name; IFNAMSIZ on Linux, including the '\0'. */
#define DEV_NAME_LEN 16

/** An address block in CIDR notation: an address and a prefix length. */
typedef struct {
  uint32_t addr;  /**< any address in the block, host byte order */
  uint8_t prefix; /**< 0 to 32 */
} cidr_t;

/** One entry in a forwarding table. */
typedef struct {
  cidr_t dest;            /**< destination block; host bits must be zero */
  uint32_t gateway;       /**< next hop, or 0 when the block is directly connected */

#include "lab.h"

#include <errno.h>

static void print_usage(FILE *out)
{
  fprintf(out,
          "Usage: myapp info <address>/<prefix>\n"
          "       myapp split <network>/<prefix> <count>\n"
          "       myapp lookup <table-file> <address>...\n");
}

static bool parse_count(const char *text, uint32_t *count)
{
  uint64_t value = 0;
  const char *cursor = text;

  if (text == NULL || count == NULL || *text == '\0' ||
      (*text == '0' && text[1] != '\0')) {
    return false;
  }
  while (*cursor != '\0') {
    if (*cursor < '0' || *cursor > '9') {
      return false;
    }
    value = value * 10U + (uint64_t)(*cursor - '0');
    if (value > UINT32_MAX) {
      return false;
    }
    cursor++;
  }
  *count = (uint32_t)value;
  return true;
}

static int command_info(const char *text)
{
  cidr_t block;
  char address[IP_STR_LEN];
  char netmask[IP_STR_LEN];
  char network[IP_STR_LEN];
  char broadcast[IP_STR_LEN];
  char first_host[IP_STR_LEN];
  char last_host[IP_STR_LEN];

  if (!cidr_parse(text, &block)) {
    fprintf(stderr, "myapp: %s: not a valid CIDR block\n", text);
    return 1;
  }

  ip_format(block.addr, address);
  ip_format(cidr_mask(block.prefix), netmask);
  ip_format(cidr_network(block), network);
  ip_format(cidr_broadcast(block), broadcast);
  ip_format(cidr_first_host(block), first_host);
  ip_format(cidr_last_host(block), last_host);
  printf("address    %s\n"
         "prefix     /%u\n"
         "netmask    %s\n"
         "network    %s\n"
         "broadcast  %s\n"
         "first host %s\n"
         "last host  %s\n"
         "hosts      %" PRIu64 "\n",
         address, (unsigned int)(block.prefix > 32 ? 32 : block.prefix),
         netmask, network, broadcast, first_host, last_host,
         cidr_host_count(block));
  return 0;
}

static int command_split(const char *block_text, const char *count_text)
{
  cidr_t block;
  cidr_t subnet;
  uint32_t count;

  if (!cidr_parse(block_text, &block) || !cidr_is_network(block) ||
      !parse_count(count_text, &count) || count == 0 ||
      (count & (count - 1U)) != 0) {
    fprintf(stderr, "myapp: invalid network or subnet count\n");
    return 1;
  }

  for (uint32_t index = 0; index < count; index++) {
    if (!cidr_subnet(block, count, index, &subnet)) {
      fprintf(stderr, "myapp: cannot split network into %u subnets\n", count);
      return 1;
    }
    char formatted[CIDR_STR_LEN];
    cidr_format(subnet, formatted);
    puts(formatted);
  }
  return 0;
}

static int command_lookup(const char *filename, int address_count,
                          char *const addresses[])
{
  FILE *in = fopen(filename, "r");
  fwd_table_t *table;
  size_t bad_line = 0;
  bool malformed = false;
  bool unreachable = false;

  if (in == NULL) {
    fprintf(stderr, "myapp: %s: cannot open table file: %s\n",
            filename, strerror(errno));
    return 2;
  }

  table = table_create();
  if (table == NULL) {
    fclose(in);
    fprintf(stderr, "myapp: could not allocate forwarding table\n");
    return 2;
  }
  if (!table_load(table, in, &bad_line)) {
    fclose(in);
    table_destroy(table);
    if (bad_line != 0) {
      fprintf(stderr, "myapp: %s:%zu: not a valid route\n",
              filename, bad_line);
    } else {
      fprintf(stderr, "myapp: %s: could not read table file\n", filename);
    }
    return 2;
  }
  fclose(in);

  for (int index = 0; index < address_count; index++) {
    uint32_t address;
    char formatted[IP_STR_LEN];
    const route_t *route;

    if (!ip_parse(addresses[index], &address)) {
      fprintf(stderr, "myapp: %s: not a valid address\n", addresses[index]);
      malformed = true;
      continue;
    }
    route = table_lookup(table, address);
    ip_format(address, formatted);
    if (route == NULL) {
      printf("%s unreachable\n", formatted);
      unreachable = true;
    } else if (route->gateway == 0) {
      printf("%s dev %s\n", formatted, route->dev);
    } else {
      char gateway[IP_STR_LEN];
      ip_format(route->gateway, gateway);
      printf("%s via %s dev %s\n", formatted, gateway, route->dev);
    }
  }

  table_destroy(table);
  return malformed ? 1 : (unreachable ? 3 : 0);
}

int main(int argc, char **argv)
{
  if (argc == 1) {
    print_usage(stdout);
    return 0;
  }
  if (argc >= 2 && strcmp(argv[1], "info") == 0 && argc == 3) {
    return command_info(argv[2]);
  }
  if (argc >= 2 && strcmp(argv[1], "split") == 0 && argc == 4) {
    return command_split(argv[2], argv[3]);
  }
  if (argc >= 3 && strcmp(argv[1], "lookup") == 0 && argc >= 4) {
    return command_lookup(argv[2], argc - 3, &argv[3]);
  }

  fprintf(stderr, "myapp: invalid command line\n");
  print_usage(stderr);
  return 1;
}
